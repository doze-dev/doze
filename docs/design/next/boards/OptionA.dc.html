# Already installed (brew, apt, nix): the command and a port
service "mailpit" {
run = "mailpit --smtp 127.0.0.1:$PORT"
port = 1025
}
# Not installed: add where to download it
service "mongo" {
download = "https://fastdl.mongodb.org/osx/mongodb-macos-arm64-8.0.4.tgz"
run = "bin/mongod --dbpath $DATA --port $PORT"
port = 27017
}
# In a container: still just your command
service "search" {
run = "docker run --rm -p $PORT:7700 getmeili/meilisearch:v1"
port = 7700
}
# Using them: plain HCL, nothing to export first
app "api" {
run = "go run ./cmd/api"
env = {
MONGO_URL = "mongodb://${service.mongo.address}/catalog"
SMTP_ADDR = service.mailpit.address
SEARCH = "http://${service.search.address}"
}
}
